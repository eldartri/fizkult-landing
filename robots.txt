# Политика простая: пускаем всех. Внутренние файлы репозитория закрыты не
# здесь, а правилами в `_redirects` (robots.txt — просьба, редирект —
# запрет); строки Disallow ниже стоят для тех роботов, которые читают
# robots.txt раньше, чем получают редирект.
#
# Новый каталог с внутренним содержимым — строка здесь И строка в
# `_redirects`. Новая публичная страница — строка в `sitemap.xml`.

# Яндекс читает ТОЛЬКО свою секцию, если она есть, — поэтому запреты
# повторены. Clean-param склеивает адреса с метками кампаний в один:
# без него /tarify?utm_source=tg живёт в индексе отдельной страницей.
User-agent: Yandex
Allow: /
Disallow: /drafts/
Disallow: /source/
Disallow: /scripts/
Clean-param: utm_source&utm_medium&utm_campaign&utm_content&utm_term&ysclid&gclid&fbclid
